## Key interfaces to define early (share between frontend/backend via types)

Consider creating `apps/server/src/types.ts` and importing in frontend via API response shapes:

```typescript
// Room state (mirrored on frontend in useRoom hook)
interface RoomUser {
  userId: string;
  name: string;
  ready: boolean;
  connected: boolean;
}

interface RoomState {
  roomId: string;
  users: RoomUser[];
  bothReady: boolean;
  canPlay: boolean;
}

// Watchlist item
interface WatchlistItem {
  id: string;
  tmdbId: string;
  imdbId: string;
  mediaType: 'movie' | 'tv';
  title: string;
  posterPath: string | null;
  addedAt: string;
  progress: WatchProgress[];
}

// Watch progress
interface WatchProgress {
  id: string;
  season: number | null;
  episode: number | null;
  watchedAt: string;
  completedBy: string[];
}
```
